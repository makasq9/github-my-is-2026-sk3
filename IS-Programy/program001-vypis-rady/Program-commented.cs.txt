// Proměnná říká, jestli se má program opakovat. Na začátku je "a", aby cyklus poprvé proběhl.
string again = "a";

// ToLower() změní písmena na malá, takže funguje "a" i "A".
while (again.ToLower() == "a")
{
    // Clear smaže starý výpis, aby při opakování nebyl chaos na obrazovce.
    Console.Clear();
    Console.WriteLine("======================================");
    Console.WriteLine("   PROGRAM 001: VÝPIS ŘADY ČÍSEL");
    Console.WriteLine("======================================");
    Console.WriteLine();

    // ReadLine() může vrátit null, proto je tam ?? "1" jako náhradní hodnota.
    // Pozor: int.Parse spadne, když uživatel napíše text.
    Console.Write("Zadejte první hodnotu (first): ");
    int first = int.Parse(Console.ReadLine() ?? "1");

    Console.Write("Zadejte poslední hodnotu (last): ");
    int last = int.Parse(Console.ReadLine() ?? "10");

    Console.Write("Zadejte krok (step): ");
    int step = int.Parse(Console.ReadLine() ?? "1");

    Console.WriteLine();
    Console.WriteLine("--- Výstup řady ---");

    // Krok musí být kladný. Při 0 nebo záporném kroku by cyklus nikdy neskončil.
    if (step > 0)
    {
        for (int i = first; i <= last; i += step)
        {
            Console.WriteLine(i);
        }
    }
    else
    {
        Console.WriteLine("Krok musí být větší než 0!");
    }

    Console.WriteLine();
    Console.WriteLine("Pro opakování programu stiskněte klávesu a.");
    Console.WriteLine("Následně potvrďte klávesou Enter.");
    Console.WriteLine("Jakoukoli jinou klávesou program ukončíte.");
    Console.WriteLine();
    Console.Write("Chcete program spustit znovu? ");
    again = Console.ReadLine() ?? "";
}
