// Pomocná funkce: ptá se na číslo tak dlouho, dokud uživatel nezadá platné.
// TryParse nevyhodí chybu, jen vrátí true/false, proto je bezpečnější než Parse.
static int ReadInt(string prompt)
{
    while (true)
    {
        Console.Write(prompt);
        if (int.TryParse(Console.ReadLine(), out int value))
        {
            return value;
        }
        Console.WriteLine("Neplatné číslo, zkuste to znovu.");
    }
}

string again = "a";

while (again.Trim().ToLower() == "a")
{
    Console.Clear();
    Console.WriteLine("======================================");
    Console.WriteLine("   PROGRAM 001: VÝPIS ŘADY ČÍSEL");
    Console.WriteLine("======================================");
    Console.WriteLine();

    int first = ReadInt("Zadejte první hodnotu (first): ");
    int last = ReadInt("Zadejte poslední hodnotu (last): ");

    // Krok se ptáme znovu, dokud není větší než 0.
    int step = ReadInt("Zadejte krok (step): ");
    while (step <= 0)
    {
        Console.WriteLine("Krok musí být větší než 0!");
        step = ReadInt("Zadejte krok (step): ");
    }

    Console.WriteLine();
    Console.WriteLine("--- Výstup řady ---");

    for (int i = first; i <= last; i += step)
    {
        Console.WriteLine(i);
    }

    Console.WriteLine();
    Console.WriteLine("Pro opakování programu stiskněte klávesu a.");
    Console.WriteLine("Následně potvrďte klávesou Enter.");
    Console.WriteLine("Jakoukoli jinou klávesou program ukončíte.");
    Console.WriteLine();
    Console.Write("Chcete program spustit znovu? ");
    again = Console.ReadLine() ?? "";
}
